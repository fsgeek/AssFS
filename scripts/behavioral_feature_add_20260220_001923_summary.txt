interface RateLimitOptions {
  windowMs: number;
  max: number;
  message?: string;
  skipSuccessfulRequests?: boolean;
}

const defaultOptions: RateLimitOptions = {
  windowMs: 60 * 1000, // 1 minute
  max: 100, // 100 requests per minute
  message: 'Too many requests, please try again later.',
  skipSuccessfulRequests: false,
};

export function createRateLimit(options: Partial<RateLimitOptions> = {}) {
  const config = { ...defaultOptions, ...options };
  
  return async (req: Request, res: Response, next: NextFunction) => {
    try {
      // For authenticated routes, use userId; for public routes, use IP
      const userId = (req as any).userId;
      const identifier = userId || req.ip || 'anonymous';
      
      // Create a unique key for this user/IP in this time window
      const windowStart = Math.floor(Date.now() / config.windowMs);
      const key = `rate_limit:${identifier}:${windowStart}`;
      
      // Get current count and increment
      const current = await cache.incr(key);
      
      // Set expiration on first request in this window
      if (current === 1) {
        await cache.expire(key, Math.ceil(config.windowMs / 1000));
      }
      
      // Add rate limit headers
      res.set({
        'X-RateLimit-Limit': config.max.toString(),
        'X-RateLimit-Remaining': Math.max(0, config.max - current).toString(),
        'X-RateLimit-Reset': new Date(
          (windowStart + 1) * config.windowMs
        ).toISOString(),
      });
      
      // Check if limit exceeded
      if (current > config.max) {
        return res.status(429).json({
          error: config.message,
          retryAfter: Math.ceil(config.windowMs / 1000),
        });
      }
      
      next();
    } catch (error) {
      console.error('Rate limiting error:', error);
      // On Redis errors, allow the request through (fail open)
      next();
    }
  };
}

```typescript
import { Request, Response, NextFunction } from 'express';
import { cache } from '../lib/cache';

const RATE_LIMIT_WINDOW = 60; // 1 minute in seconds
const RATE_LIMIT_MAX_REQUESTS = 100;

export async function rateLimit(req: Request, res: Response, next: NextFunction) {
  try {
    // Determine identifier: authenticated user ID or IP address
    const userId = (req as any).userId;
    const identifier = userId || req.ip || req.socket.remoteAddress || 'unknown';
    const key = `rate-limit:${identifier}`;

    // Increment counter atomically
    const currentCount = await cache.incr(key);

    // Set expiration on first request in window
    if (currentCount === 1) {
      await cache.expire(key, RATE_LIMIT_WINDOW);
    }

    // Add rate limit headers to response
    res.setHeader('X-RateLimit-Limit', RATE_LIMIT_MAX_REQUESTS);
    res.setHeader('X-RateLimit-Remaining', Math.max(0, RATE_LIMIT_MAX_REQUESTS - currentCount));
    res.setHeader('X-RateLimit-Reset', Math.floor(Date.now() / 1000) + RATE_LIMIT_WINDOW);

    // Check if limit exceeded
    if (currentCount > RATE_LIMIT_MAX_REQUESTS) {
      return res.status(429).json({
        error: 'Too Many Requests',
        message: `Rate limit exceeded: ${RATE_LIMIT_MAX_REQUESTS} requests per ${RATE_LIMIT_WINDOW} seconds`,
        retryAfter: RATE_LIMIT_WINDOW,
      });
    }

    next();
  } catch (error) {
    // On Redis error, fail open (allow request) to prevent service degradation
    console.error('Rate limit check failed:', error);
    next();
  }
}
```

**Why this approach:**
- Uses Redis INCR for atomic operation (thread-safe across load balancer instances)
- EXPIRE ensures old keys are cleaned up automatically
- Fails open on Redis errors (requests still processed)
- Per-user tracking for authenticated users, per-IP for anonymous
- Standard HTTP rate limit headers (X-RateLimit-*)
